cmake_minimum_required(VERSION 3.24)
project(AstroImageAnalyzer VERSION 0.2.0 LANGUAGES C CXX)
option(ASTRO_FETCH_OPENCV "Build pinned minimal OpenCV (requires network on first configure)" OFF)
if(ASTRO_FETCH_OPENCV)
  include(FetchContent)
  set(BUILD_LIST core,imgproc,imgcodecs CACHE STRING "" FORCE)
  set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
  foreach(flag BUILD_TESTS BUILD_PERF_TESTS BUILD_EXAMPLES BUILD_opencv_apps BUILD_JAVA BUILD_opencv_python3 WITH_IPP WITH_OPENCL WITH_ITT WITH_FFMPEG WITH_GSTREAMER WITH_OPENEXR WITH_WEBP WITH_JASPER WITH_OPENJPEG WITH_AVIF WITH_JPEGXL WITH_TBB WITH_OPENMP)
    set(${flag} OFF CACHE BOOL "" FORCE)
  endforeach()
  foreach(codec JPEG PNG TIFF ZLIB)
    set(BUILD_${codec} ON CACHE BOOL "" FORCE)
  endforeach()
  set(CPU_DISPATCH "" CACHE STRING "" FORCE)
  set(BUILD_WITH_STATIC_CRT ON CACHE BOOL "" FORCE)
  FetchContent_Declare(opencv URL https://github.com/opencv/opencv/archive/refs/tags/4.12.0.tar.gz
    URL_HASH SHA256=44c106d5bb47efec04e531fd93008b3fcd1d27138985c5baf4eafac0e1ec9e9d)
  FetchContent_MakeAvailable(opencv)
  # Match the static runtime selected by the bundled OpenCV build on MSVC.
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  set(OpenCV_LIBS opencv_core opencv_imgproc opencv_imgcodecs)
  set(OpenCV_INCLUDE_DIRS ${opencv_SOURCE_DIR}/modules/core/include ${opencv_SOURCE_DIR}/modules/imgproc/include ${opencv_SOURCE_DIR}/modules/imgcodecs/include ${opencv_BINARY_DIR} ${CMAKE_BINARY_DIR})
else()
  find_package(OpenCV 4 REQUIRED COMPONENTS core imgproc imgcodecs)
endif()
# OpenCV sets a global bin directory; keep this project's CLI location stable.
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
add_library(astro_core src/analysis.cpp src/quality.cpp src/pipeline.cpp)
target_include_directories(astro_core PUBLIC include)
target_compile_features(astro_core PUBLIC cxx_std_17)
add_library(astro_source src/file_source.cpp)
target_link_libraries(astro_source PUBLIC astro_core PRIVATE ${OpenCV_LIBS})
target_include_directories(astro_source SYSTEM PRIVATE ${OpenCV_INCLUDE_DIRS})
add_executable(astro_analyze src/main.cpp)
target_link_libraries(astro_analyze PRIVATE astro_source)
if(MSVC)
  target_compile_options(astro_core PRIVATE /W4 /utf-8)
  target_compile_options(astro_source PRIVATE /W4 /utf-8)
  target_compile_options(astro_analyze PRIVATE /W4 /utf-8)
else()
  target_compile_options(astro_core PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(astro_source PRIVATE -Wall -Wextra -Wpedantic)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(astro_tests tests/tests.cpp)
  target_link_libraries(astro_tests PRIVATE astro_source ${OpenCV_LIBS})
  target_include_directories(astro_tests SYSTEM PRIVATE ${OpenCV_INCLUDE_DIRS})
  add_test(NAME astro_focused COMMAND astro_tests)
  add_executable(astro_quality_tests tests/quality_tests.cpp)
  target_link_libraries(astro_quality_tests PRIVATE astro_core)
  add_test(NAME astro_quality COMMAND astro_quality_tests)
  add_test(NAME astro_cli COMMAND ${CMAKE_COMMAND}
    -DCLI=$<TARGET_FILE:astro_analyze> -DWORK=${CMAKE_CURRENT_BINARY_DIR}/cli-tests
    -P ${CMAKE_CURRENT_SOURCE_DIR}/tests/cli.cmake)
endif()
